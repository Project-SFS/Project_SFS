CREATE TABLE submissions (ID INT IDENTITY(1,1) PRIMARY KEY, PROBLEM_ID INT, TEAM_EMAIL VARCHAR(100), SOL_TITLE VARCHAR(100), SOL_DESCRIPTION TEXT, SUB_DATE DATE, STATUS VARCHAR(20) DEFAULT 'PENDING', SOL_LINK VARCHAR(256), FILES VARCHAR(256), MARK INT, CP_MARK INT, PS_MARK INT, BV_MARK INT, FP_MARK INT, IN_MARK INT);


CREATE TABLE problems (ID INT IDENTITY(1,1) PRIMARY KEY, TITLE VARCHAR(100), DESCRIPTION TEXT, SUB_DEADLINE DATE, CATEGORY VARCHAR(50), DEPT VARCHAR(50), Reference VARCHAR(256), Evaluator_ID INT);

CREATE TABLE Team_Members_List (ID INT IDENTITY(1,1) PRIMARY KEY, NAME VARCHAR(50), SPOC_ID INT, EMAIL VARCHAR(100), PHONE VARCHAR(20), GENDER VARCHAR(10), ROLE VARCHAR(20), Team_ID INT);

CREATE TABLE Team_List (ID INT IDENTITY(1,1) PRIMARY KEY, NAME VARCHAR(50), SPOC_ID INT, LEAD_EMAIL VARCHAR(50), LEAD_PHONE VARCHAR(50), MENTOR_NAME VARCHAR(50), MENTOR_EMAIL VARCHAR(50));
